\begin{worked}{Geometric Distribution MAP Parameter Estimation}
Geometric likelihood $P(Y=y \mid X=x) = x(1-x)^{y-1}$ with prior $P(X=x) = 2x$. Given $Y = 3$:
\begin{steps}
  \item $g(x) = 2x^2(1-x)^2 \implies \frac{d\ln g(x)}{dx} = \frac{2}{x} - \frac{2}{1-x} = 0 \implies x_{\text{MAP}} = 0.5$.
\end{steps}
\end{worked}
